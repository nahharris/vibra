use @std.fs.DirReadText enduse

use @std.fs.FsError enduse

type ReaderResult
  case absent
  case present {reader: DirReadText}
endtype

proc readOptional
  takes {reader: Option(DirReadText) at: RelativePath fallback: Str}
  returns text: Result(Str FsError)
  performs {DirReadText}
  visibility @public
  do
    return match reader
      case Option.none()
        Result.ok(value: fallback)
      case Option.some(value: grant)
        await DirReadText.read(grant at)
    endmatch
endproc
