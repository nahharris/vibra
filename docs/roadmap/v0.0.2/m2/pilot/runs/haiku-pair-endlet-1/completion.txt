// Orders the pair and optionally reports the difference of its ends.
fn normalizePair
  takes {{low: lowValue high: highValue}: {low: I32 high: I32}}
  labelled {includeDelta: Bool be true}
  returns pair: {first: I32 second: I32 delta: Option(I32)}
  visibility @public
  do
    let first be if Ordered.lessOrEqual(lowValue highValue) then lowValue else highValue endif endlet
    let second be if Ordered.lessOrEqual(lowValue highValue) then highValue else lowValue endif endlet
    let delta be if includeDelta then Option.some(value: I32.sub(second first)) else Option.none() endif endlet
    return {first: first second: second delta: delta}
endfn
