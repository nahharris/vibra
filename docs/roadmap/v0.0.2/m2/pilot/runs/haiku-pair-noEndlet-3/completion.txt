// Module @pilot.pair

fn normalizePair
  takes {{low: low high: high}: {low: I32 high: I32}}
  labelled {includeDelta: Bool be true}
  returns pair: {first: I32 second: I32 delta: Option(I32)}
  visibility @public
  do
    let lesser be if Ordered.lessOrEqual(low high) then low else high endif
    let greater be if Ordered.lessOrEqual(low high) then high else low endif
    let delta: Option(I32) be if includeDelta then
      Option.some(value: I32.sub(greater lesser))
    else
      Option.none()
    endif
    return {first: lesser second: greater delta: delta}
endfn
