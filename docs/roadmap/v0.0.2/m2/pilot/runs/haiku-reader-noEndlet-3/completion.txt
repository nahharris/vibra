use @std.fs.DirReadText enduse

use @std.fs.FsError enduse

use @std.fs.RelativePath enduse

proc readOptional
  takes {reader: Option(DirReadText) at: RelativePath fallback: Str}
  returns text: Result(Str FsError)
  performs {DirReadText}
  visibility @public
  do
    let Option.some(value: grant) be reader else
      return Result.ok(value: fallback)
    endlet
    return await DirReadText.read(grant at)
endproc
