type Row of {key: Str amount: I32}
  visibility @public
endtype

fn indexRows
  takes {rows: List(Row)}
  returns index: Dict(Str I32)
  visibility @public
  do
    let pairs be Iter.map(
      rows
      fn
        takes {row: Row}
        returns pair: {Str I32}
        do
          return {row.key row.amount}
      endfn
    )
    let collected be Iter.collect(pairs)
    return From.convert(collected)
endfn

fn latest
  takes {rows: List(Row) key: Str}
  returns amount: Option(I32)
  visibility @public
  do
    return indexRows(rows)(key)
endfn
